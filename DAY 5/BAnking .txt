Banking Transaction History
System processes multiple transactions.
For each transaction:

If transaction type = “withdrawal” and balance sufficient → deduct
If insufficient → reject
If transaction type = “deposit” → add to balance
Finally display final balance.

FUNCTION main()
    balance=30000
    transactions=[
        {"type": "withdrawal", "amount": 5000},
        {"type": "deposit", "amount": 10000},
        {"type": "withdrawal", "amount": 40000}
    ]
    finalBalance=process(balance, transactions)
    DISPLAY finalBalance
END FUNCTION

FUNCTION process(balance, transactions)
    For each transaction in transactions
    If transaction["type"] = "withdrawal" THEN
        IF balance >= transaction["amount"] THEN
            balance=balance-transaction["amount"]
        ELSE
            DISPLAY "Transaction rejected: Insufficient balance"
        END IF
    ELSE IF transaction["type"] = "deposit" THEN
        balance=balance+transaction["amount"]
    END IF
    End for
    Return balance
END FUNCTION
