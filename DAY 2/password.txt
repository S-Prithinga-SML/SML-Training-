A system checks the length of a password.
If length is 8 or more characters, display “Strong password”.
Otherwise, display “Weak password”.

FUNCTION main()
    password=123456789
    result=passwordcheck(password)
    DISPLAY result
    END FUNCTION

FUNCTION passwordcheck(password)
    IF LENGTH(password) >= 8 THEN
    Return "Strong password"
    ELSE
    Return "Weak password"
    END IF
    END FUNCTION
